## 2. ОБНОВЛЁННЫЙ СКРИПТ: `WFS_CORE/find_normalization.py`

Ищет маркеры **только в `WFS_CORE/`** (со всеми подпапками), выводит в папку архетипа.

```python
#!/usr/bin/env python3
"""
Поиск маркеров нормализации архетипов #WFS.
Запуск: python3 WFS_CORE/find_normalization.py <archetype_key>
"""

import os
import sys
import re
from datetime import datetime

# ─── МАРКЕРЫ НОРМАЛИЗАЦИИ ПО АРХЕТИПАМ ───
NORMALIZATION_PATTERNS = {
    "tufyak": [
        "это нормально", "так всегда было", "так сложилось", "человеческий фактор",
        "все так делают", "ничего страшного", "само рассосётся", "само пройдёт",
        "я не виноват", "так получилось", "судьба", "карма", "воля Бога",
        "не суетитесь", "смирись", "терпи", "я грешный", "потом", "нет сил",
        "устал", "не могу", "бесполезно", "всё равно ничего не изменится",
        "зачем напрягаться", "и так сойдёт", "переждём", "отложим",
        "не сейчас", "позже", "завтра", "авось", "как-нибудь", "само собой"
    ],
    # Следующие архетипы добавим по мере работы.
    # "kisel": [...],
    # "straus": [...],
    # "fazan": [...],
}

# ─── ЕДИНАЯ ПАПКА ПОИСКА ───
SEARCH_ROOT = "WFS_CORE"

OUTPUT_BASE = "WFS_CORE/КНИГИ/##АРХЕТИПЫ_СВОЙСТВА"
CHUNK_SIZE = 500

SS_METRIC_PATTERN = re.compile(r'\[(#[A-ZА-Я_]+):\s*([^\]]+)\]')
SS_AGENT_PATTERN = re.compile(r'\[ИМЯ:\s*([^\]]+)\]')


def extract_metrics(ss_line):
    metrics = {}
    for m in SS_METRIC_PATTERN.finditer(ss_line):
        metrics[m.group(1).strip()] = m.group(2).strip()
    return metrics


def extract_agent(ss_line):
    m = SS_AGENT_PATTERN.search(ss_line)
    return m.group(1).strip() if m else None


def find_nearest_ss(lines, idx, max_dist=50):
    for i in range(idx, max(-1, idx - max_dist), -1):
        if '[ИМЯ:' in lines[i]:
            return lines[i].strip()
    for i in range(idx, min(len(lines), idx + max_dist)):
        if '[ИМЯ:' in lines[i]:
            return lines[i].strip()
    return None


def find_normalization(archetype_key):
    if archetype_key not in NORMALIZATION_PATTERNS:
        print(f"Ошибка: маркеры для '{archetype_key}' не заданы.")
        print(f"Доступные: {', '.join(NORMALIZATION_PATTERNS.keys())}")
        return

    markers = NORMALIZATION_PATTERNS[archetype_key]
    results = []
    file_count = 0
    seen_quotes = set()

    for root, dirs, files in os.walk(SEARCH_ROOT):
        for fname in files:
            if not fname.endswith(('.md', '.txt')):
                continue
            fpath = os.path.join(root, fname)
            file_count += 1
            try:
                with open(fpath, 'r', encoding='utf-8', errors='replace') as f:
                    lines = f.readlines()
            except Exception:
                continue

            for i, line in enumerate(lines):
                for marker in markers:
                    if marker.lower() in line.lower():
                        start = max(0, i - 4)
                        end = min(len(lines), i + 5)
                        context = ''.join(lines[start:end]).strip()
                        norm = context[:200].lower()
                        if norm in seen_quotes:
                            break
                        seen_quotes.add(norm)

                        ss_line = find_nearest_ss(lines, i)
                        metrics = extract_metrics(ss_line) if ss_line else {}
                        agent = extract_agent(ss_line) if ss_line else None

                        results.append({
                            'file': fpath,
                            'line': i + 1,
                            'marker': marker,
                            'quote': context,
                            'agent': agent,
                            'metrics': metrics,
                        })
                        break

    folder_name = f"{archetype_key.upper()}_NORM"
    out_dir = os.path.join(OUTPUT_BASE, folder_name)
    os.makedirs(out_dir, exist_ok=True)

    if not results:
        single_file = os.path.join(out_dir, f"{folder_name}_001.md")
        with open(single_file, 'w', encoding='utf-8') as f:
            f.write(f"# NORMALIZATION DATA: {folder_name}\n")
            f.write(f"**Дата:** {datetime.now().isoformat()}\n")
            f.write(f"**Найдено:** 0 совпадений\n\n")
            f.write("NO DATA\n")
        print(f"Нет данных. Создан пустой файл: {single_file}")
        return

    total_parts = (len(results) + CHUNK_SIZE - 1) // CHUNK_SIZE

    for part_idx in range(total_parts):
        start = part_idx * CHUNK_SIZE
        end = start + CHUNK_SIZE
        chunk = results[start:end]

        part_num = f"{part_idx + 1:03d}"
        part_file = os.path.join(out_dir, f"{folder_name}_{part_num}.md")

        with open(part_file, 'w', encoding='utf-8') as f:
            f.write(f"# NORMALIZATION DATA: {folder_name}\n")
            f.write(f"**Дата:** {datetime.now().isoformat()}\n")
            f.write(f"**Часть:** {part_num} из {total_parts:03d}\n")
            f.write(f"**Записей в части:** {len(chunk)}\n")
            f.write(f"**Всего записей:** {len(results)}\n")
            f.write(f"**Проверено файлов:** {file_count}\n\n")

            for r in chunk:
                f.write(f"---\n")
                f.write(f"**[FILE]:** `{r['file']}`\n")
                f.write(f"**[LINE]:** {r['line']}\n")
                f.write(f"**[MARKER]:** {r['marker']}\n")
                f.write(f"**[AGENT]:** {r['agent'] or 'N/A'}\n")
                if r['metrics']:
                    metrics_str = ' '.join(f"{k}={v}" for k, v in r['metrics'].items())
                    f.write(f"**[METRICS]:** {metrics_str}\n")
                else:
                    f.write(f"**[METRICS]:** N/A\n")
                f.write(f"**[QUOTE]:**\n```\n{r['quote']}\n```\n\n")

    print(f"Готово.")
    print(f"Папка: {out_dir}/")
    print(f"Частей: {total_parts}")
    print(f"Всего записей: {len(results)} (проверено файлов: {file_count})")


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Использование: python3 WFS_CORE/find_normalization.py <archetype>")
        print(f"Доступные: {', '.join(NORMALIZATION_PATTERNS.keys())}")
    else:
        find_normalization(sys.argv[1].lower())
```
